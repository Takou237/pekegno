\section{Répartition des Responsabilités}

\subsection*{Chef de Projet (Topo Boss)}
\begin{itemize}
    \item Valide les specs et priorise les modules
    \item Recette après chaque module (validation avant de passer au suivant)
    \item Décide des ajustements en cours de route
    \item Fournit les accès et informations nécessaires
\end{itemize}

\subsection*{Dev1 (Fullstack)}
\begin{itemize}
    \item Backend : Modélisation BDD, API Laravel, migrations, seeders
    \item Frontend : Composants React, pages, intégration API
    \item \textbf{Lead technique} : Architecture, revue de code, décisions tech
    \item Modules principaux : Agences, Services, Formations, Facturation, Créances
\end{itemize}

\subsection*{Dev2 (Fullstack)}
\begin{itemize}
    \item Backend : API Laravel, intégrations paiement, notifications, jobs
    \item Frontend : Composants React, pages, tableaux de bord
    \item \textbf{Lead fonctionnel} : CRM, Commercial, Comptabilité, Rapports
    \item Modules principaux : CRM, Ventes, Clients, Comptabilité, Tableaux de bord
\end{itemize}

\section{Méthodologie de Travail}

\begin{mdframed}[backgroundcolor=lightgray!30, linecolor=primary]
    \textbf{Processus par module :}
    Dev1 écrit les migrations + seeders $\rightarrow$ Dev1 écrit l'API $\rightarrow$ Dev2 écrit le frontend $\rightarrow$ Tests $\rightarrow$ Recette Topo Boss
\end{mdframed}

\subsection*{Règles d'Or}
\begin{enumerate}
    \item \textbf{Un module à la fois} -- on termine complètement un module avant de commencer le suivant
    \item \textbf{Tests avant recette} -- Dev1 teste l'API, Dev2 teste le frontend, puis validation croisée
    \item \textbf{Commit quotidien} -- fin de journée, tout ce qui est fonctionnel est commité
    \item \textbf{Point journalier (15 min)} -- chaque matin, synchronisation Dev1 + Dev2
    \item \textbf{Recette Topo Boss} -- validation obligatoire avant de passer au module suivant
\end{enumerate}

\section{Processus de Recette par Module}

Chaque module suit ce cycle avant de passer au suivant :

\begin{mdframed}[backgroundcolor=lightgray!20, linecolor=primary]
    \textbf{Étape 1 :} Dev1 implémente le backend (migration + API + tests) \\
    \textbf{Étape 2 :} Dev2 implémente le frontend (pages + composants) \\
    \textbf{Étape 3 :} Test croisé (Dev1 teste le frontend, Dev2 teste l'API) \\
    \textbf{Étape 4 :} Démo au Chef de Projet (Topo Boss) \\
    \textbf{Étape 5 :} Validation ou retour pour ajustements \\
    \textbf{Étape 6 :} Passage au module suivant
\end{mdframed}

\section{Checklist de Validation}

\subsection*{Pour chaque module, vérifier :}
\begin{itemize}
    \item CRUD complet fonctionnel (créer, lire, modifier, supprimer/archiver)
    \item Permissions respectées (rôles et privilèges)
    \item Recherche, filtres et pagination opérationnels
    \item Export (Excel/PDF) si applicable
    \item Temps de réponse < 3 secondes
    \item Responsive (mobile, tablette, desktop)
    \item Pas de régression sur les modules précédents
\end{itemize}
